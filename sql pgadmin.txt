
---table creation users---
create table users (user_id serial primary key,username varchar(90) not null,email varchar(60) not null unique,password_hash text,role varchar(20),is_active boolean,created_at timestamp not null default current_timestamp,updated_at timestamp not null default current_timestamp);
---customers--
create table customers(customer_id serial primary key,user_id int not null unique, full_name varchar(100) not null,date_of_birth date not null,phone varchar(15) not null unique,email varchar(100) not null unique,address text,status varchar(20) not null default 'ACTIVE' check(status in ('ACTIVE','INACTIVATE')),created_at timestamp not null default current_timestamp,updated_at timestamp not null default current_timestamp ,FOREIGN KEY (user_id) references users(user_id));

---fd---
create table fixed_deposits(fd_id serial primary key,customer_id int not null,principal_amount numeric(15,2) not null check(principal_amount > 0),interest_rate numeric(15,2) not null check (interest_rate >0),tenure_months int not null check(tenure_months>0),start_date date not null,maturity_date date not null,maturity_amount numeric(15,2) not null check (maturity_amount>0),status varchar(30) default 'ACTIVE' check(status in ('ACTIVE','MATURED','PREMATURE-CLOSED','CLOSED')),created_at timestamp default current_timestamp,updated_at timestamp default current_timestamp,constraint fk foreign key (customer_id) references customers(customer_id),constraint chkfd check(maturity_date>start_date)); 

----fdt----
create table fixed_transactions(transaction_id serial primary key,fd_id int not null unique, transaction_type varchar(40) check(transaction_type in ('FD_CREATED','INTEREST_CREATED','MATURITY_PROCESSED','PREMATURE_WITHDRAWAL','FD_CLOSED')),amount numeric(15,2) not null check(amount>=0),transaction_date timestamp not null default current_timestamp,reference_number varchar(50) not null unique,description text,created_by int not null,constraint fkd foreign key (fd_id) references fixed_deposits(fd_id),constraint fkt foreign key (created_by) references users(user_id));
---audit---
create table audit_logs(audit_id serial primary key,entity_type varchar(50) not null,entity_id int not null,old_data jsonb,new_data jsonb,performed_by int,created_at timestamp default current_timestamp,constraint fka foreign key(performed_by) references users(user_id));
